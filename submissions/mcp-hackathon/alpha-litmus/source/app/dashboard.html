REST call

{
  "schema_version": "alphalitmus-release-gate-1",
  "decision": "BLOCK_DEPLOYMENT | INSUFFICIENT_EVIDENCE | SURVIVED_BOUNDED_TESTS",

  "profitability_claimed": false
}
